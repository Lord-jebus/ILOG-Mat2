El Hardware: Arduino UNO – ATmega328P

Características del ATmega328P
  • CPU: 8 bits, arquitectura RISC
  • Frecuencia: 16 MHz (Arduino UNO)
  • Flash: 32 KB (programa)
  • SRAM: 2 KB (variables en tiempo de ejecución)
  • EEPROM: 1 KB (datos persistentes)
  • ADC: 10 bits, 6 canales analógicos
  • PWM: 6 canales (Timers 0, 1, 2)
Restricciones críticas
  • Sin FPU: float = ~15-80 ciclos por operación
  • 2 KB SRAM: la tabla de reglas + MFs + pila NO puede exceder esto
  • Bucle de control objetivo: < 10 ms (100 Hz)
Tiempo de operaciones
Suma int16:~2 ciclos
Mult int16:~2 ciclos
Div int16:~40 ciclos
Float add:~15 ciclos
Float mult:~15 ciclos
expf():~200 ciclos

¿Float o Entero? La Decisión de Representación

Opción A: Punto Flotante (float)
  • $\mu \in [0.0, 1.0]$ directamente
  • Fácil de leer y depurar
  • Exactitud: ~7 decimales significativos
  • Desventaja: lento sin FPU, ocupa 4 bytes
float mu = (x - a) / (b - a); // <- ~40 ciclos
Opción B: Punto Fijo (int16_t)
  • $\mu \times 1000 \in [0, 1000]$ como entero
  • Operaciones en 2-4 ciclos
  • Ocupa 2 bytes, cabe en registros
  • Desventaja: cuidar desbordamiento
int16_t mu = ((x-a)*1000)/(b-a); // <- ~8 ciclos
Para este lab

Implementaremos con float primero (correctitud), luego optimizaremos con int16_t (velocidad) en Clase 21.

Representación de MFs con struct

/**
 * fuzzy_types.h  -  Tipos de datos para Control Difuso
 * Clase 8 | AVR-GCC / Arduino UNO
 */
#ifndef FUZZY_TYPES_H
#define FUZZY_TYPES_H

// ----- Funciones de pertenencia -----
typedef struct {
    float a, b, c;          // pie-izq, pico, pie-der
} TriMF;

typedef struct {
    float a, b, c, d;       // flancos izq. y der.
} TrapMF;

// ----- Resultado de fuzzificacion -----
#define N_TERMS 5

typedef struct {
    float   mu[N_TERMS];    // grados de pertenencia
    uint8_t n;              // numero de terminos activos
} FuzzyDegrees;

// ----- Tipos de MF -----
#define MF_TRAP_LEFT  0     // trapezoidal izquierdo
#define MF_TRIANGLE   1     // triangular
#define MF_TRAP_RIGHT 2     // trapezoidal derecho

#endif // FUZZY_TYPES_H

Implementación de trimf y trapmf

#ifndef FUZZY_MF_H
#define FUZZY_MF_H
#include "fuzzy_types.h"

// ----- MF Triangular -----
inline float trimf(float x, const TriMF& mf) {
    if (x <= mf.a || x >= mf.c) return 0.0f;
    if (x <= mf.b) return (x - mf.a) / (mf.b - mf.a);
    return               (mf.c - x) / (mf.c - mf.b);
}

// ----- MF Trapezoidal izquierda (para NB, NN) -----
inline float ltrap(float x, float b, float d) {
    if (x <= b) return 1.0f;
    if (x >= d) return 0.0f;
    return (d - x) / (d - b);
}

// ----- MF Trapezoidal derecha (para PB, PP) -----
inline float rtrap(float x, float a, float c) {
    if (x <= a) return 0.0f;
    if (x >= c) return 1.0f;
    return (x - a) / (c - a);
}

// ----- Funcion min() para fuerzas de disparo -----
inline float fmin2(float a, float b) {
    return (a < b) ? a : b;
}

#endif // FUZZY_MF_H

Fuzzificador de Error de Temperatura

#include "fuzzy_mf.h"

// === Particion para Error e in [-30, +30] deg C ===
// Indices: 0=NB  1=NS  2=ZO  3=PS  4=PB
static const float E_BOUND_NB_D =  -10.0f; // ltrap: b=-30, d=-10
static const float E_BOUND_NB_B =  -30.0f;
static const TriMF E_NS = {-30.0f, -15.0f,   0.0f};
static const TrapMF E_ZO_T = {-10.0f, -5.0f, 5.0f, 10.0f};
static const TriMF E_PS = {  0.0f,  15.0f,  30.0f};
static const float E_BOUND_PB_A =   10.0f; // rtrap: a=10, c=30
static const float E_BOUND_PB_C =   30.0f;

FuzzyDegrees fuzzify_error(float e) {
    FuzzyDegrees fd;
    fd.n = N_TERMS;

    fd.mu[0] = ltrap(e, E_BOUND_NB_B, E_BOUND_NB_D); // NB
    fd.mu[1] = trimf(e, E_NS);                        // NS
    // ZO: trapezoidal plano con nucleo [-5, +5]
    fd.mu[2] = (e >= -5.0f && e <= 5.0f) ? 1.0f :
               (e >  -10.0f && e < -5.0f) ?
                   (e + 10.0f) / 5.0f :
               (e >   5.0f && e < 10.0f) ?
                   (10.0f - e) / 5.0f : 0.0f;         // ZO
    fd.mu[3] = trimf(e, E_PS);                        // PS
    fd.mu[4] = rtrap(e, E_BOUND_PB_A, E_BOUND_PB_C); // PB
    return fd;
}

Fuzzificador de Cambio de Error

#include "fuzzy_mf.h"

// === Particion para Cambio de Error de in [-5,+5] deg C/s ===
// Indices: 0=NN  1=NS  2=ZO  3=PS  4=PP
static const float DE_NN_B = -5.0f, DE_NN_D = -1.5f;
static const TriMF DE_NS   = {-5.0f,  -2.5f,  0.0f};
static const TriMF DE_ZO   = {-2.5f,   0.0f,  2.5f};
static const TriMF DE_PS   = { 0.0f,   2.5f,  5.0f};
static const float DE_PP_A =  1.5f, DE_PP_C =  5.0f;

FuzzyDegrees fuzzify_derror(float de) {
    FuzzyDegrees fd;
    fd.n = N_TERMS;

    fd.mu[0] = ltrap(de, DE_NN_B, DE_NN_D);  // NN
    fd.mu[1] = trimf(de, DE_NS);              // NS
    fd.mu[2] = trimf(de, DE_ZO);              // ZO
    fd.mu[3] = trimf(de, DE_PS);              // PS
    fd.mu[4] = rtrap(de, DE_PP_A, DE_PP_C);  // PP
    return fd;
}
Verificación rápida

Para $\dot{e} = +2$°C/s:   $\mu_{ZO} = (2.5-2)/2.5 = 0.20$   $\mu_{PS} = (2-0)/2.5 = 0.80$   $\sum = 1.00$ ✓

/**
 * fuzzy_tclab.ino
 * Fuzzificador completo para TCLAB - Arduino UNO
 * Clase 8 - Control Difuso
 *
 * Hardware: Arduino UNO + TC-Lab shield
 * Pin A0: Sensor temperatura T1 (termistor)
 * Pin 3:  PWM calentador Q1 (0-100%)
 */
#include <Arduino.h>
#include "fuzzy_types.h"
#include "fuzzy_mf.h"
#include "fuzzifier_error.h"
#include "fuzzifier_derror.h"

// === Constantes del sistema ===
const float T_REF    = 50.0f;   // Setpoint [deg C]
const float DT       = 0.1f;    // Periodo de muestreo [s]
const float ADC_REF  = 5.0f;    // Tension de referencia ADC [V]
const float TEMP_CAL = 0.0f;    // Offset de calibracion [deg C]

// === Variables globales ===
float T_prev   = 25.0f;   // Temperatura anterior [deg C]
float T_actual = 25.0f;   // Temperatura actual [deg C]
float e_prev   = 0.0f;    // Error anterior

FuzzyDegrees mu_e;    // Grados de pertenencia del error
FuzzyDegrees mu_de;   // Grados de pertenencia del cambio

unsigned long t_prev = 0; // Para medir ciclo de control

// === Nombres de terminos (para debug) ===
const char* TERMS_E[]  = {"NB","NS","ZO","PS","PB"};
const char* TERMS_DE[] = {"NN","NS","ZO","PS","PP"};

Sketch Principal: Lectura de Sensor y Loop

// === Conversion ADC -> Temperatura (termistor NTC simplificado) ===
float read_temperature() {
    int adc_raw = analogRead(A0);
    // Modelo lineal simplificado para el lab
    float voltage = adc_raw * (ADC_REF / 1023.0f);
    float temp    = (voltage - 0.5f) * 100.0f + TEMP_CAL;
    return temp;
}

void setup() {
    Serial.begin(115200);
    pinMode(3, OUTPUT);  // PWM calentador
    analogWrite(3, 0);   // Calentador apagado al inicio
    Serial.println("e[C]\tde[C/s]\tmu_NB\tmu_NS\tmu_ZO\tmu_PS\tmu_PB");
    t_prev = millis();
}

void loop() {
    // --- Esperar periodo de muestreo ---
    while ((millis() - t_prev) < (unsigned long)(DT * 1000)) {}
    t_prev = millis();

    // --- Leer sensor y calcular variables ---
    T_actual = read_temperature();
    float e  = T_REF - T_actual;
    float de = (e - e_prev) / DT;
    e_prev   = e;

    // --- FUZZIFICACION ---
    unsigned long t0 = micros();
    mu_e  = fuzzify_error(e);
    mu_de = fuzzify_derror(de);
    unsigned long t_fuzz = micros() - t0;

    // --- Imprimir resultados ---
    Serial.print(e,2);   Serial.print('\t');
    Serial.print(de,2);  Serial.print('\t');
    for (uint8_t i = 0; i < N_TERMS; i++) {
        Serial.print(mu_e.mu[i], 3); Serial.print('\t');
    }
    Serial.print("| t_fuzz="); Serial.print(t_fuzz);
    Serial.println(" us");

    T_prev = T_actual;
}

Función de Test: Verificación con Casos Conocidos

/**
 * test_fuzzifier.ino
 * Verificacion del fuzzificador contra calculos manuales
 * Ejecutar UNA VEZ en setup() antes del bucle de control
 */

struct TestCase {
    float e;          // error de entrada
    float mu_exp[5];  // mu esperados: {NB,NS,ZO,PS,PB}
    float tol;        // tolerancia admisible
};

// Casos del ejercicio resuelto en Clase 7
const TestCase TESTS[] = {
    { 7.0f, {0.000f, 0.000f, 0.300f, 0.467f, 0.000f}, 0.01f},
    {-7.0f, {0.000f, 0.467f, 0.300f, 0.000f, 0.000f}, 0.01f},
    { 0.0f, {0.000f, 0.000f, 1.000f, 0.000f, 0.000f}, 0.01f},
    {-25.0f,{1.000f, 0.333f, 0.000f, 0.000f, 0.000f}, 0.01f},
    { 25.0f,{0.000f, 0.000f, 0.000f, 0.333f, 1.000f}, 0.01f},
};

void run_tests() {
    uint8_t n_tests = sizeof(TESTS) / sizeof(TestCase);
    uint8_t passed  = 0;
    Serial.println("=== TEST FUZZIFICADOR ===");
    for (uint8_t t = 0; t < n_tests; t++) {
        FuzzyDegrees fd = fuzzify_error(TESTS[t].e);
        bool ok = true;
        for (uint8_t i = 0; i < N_TERMS; i++) {
            if (abs(fd.mu[i] - TESTS[t].mu_exp[i]) > TESTS[t].tol) {
                ok = false;
            }
        }
        Serial.print("Test e="); Serial.print(TESTS[t].e);
        Serial.println(ok ? " PASS" : " FAIL <<< ERROR");
        if (ok) passed++;
    }
    Serial.print(passed); Serial.print("/");
    Serial.print(n_tests); Serial.println(" tests passed");
}

Tabla de Reglas en Flash con PROGMEM

/**
 * rule_table.h
 * Tabla de reglas 5x5 almacenada en Flash (PROGMEM)
 * Indice salida: 0=CE 1=CB 2=M 3=AB 4=AE
 * Fila=error(NB..PB), Col=d_error(NN..PP)
 */
#ifndef RULE_TABLE_H
#define RULE_TABLE_H
#include <avr/pgmspace.h>

#define N_RULES_E  5
#define N_RULES_DE 5

// Tabla [error][d_error]
const uint8_t RULE_TABLE[N_RULES_E][N_RULES_DE] PROGMEM = {
    // NN  NS  ZO  PS  PP    <- d_error
    {  4,  4,  3,  3,  2 }, // NB (error)
    {  4,  3,  3,  2,  1 }, // NS
    {  3,  3,  2,  1,  1 }, // ZO
    {  2,  1,  1,  1,  0 }, // PS
    {  2,  1,  1,  0,  0 }, // PB
};

// Acceso seguro a la tabla
inline uint8_t get_rule(uint8_t i_e, uint8_t i_de) {
    return pgm_read_byte(&RULE_TABLE[i_e][i_de]);
}

#endif // RULE_TABLE_H
Ahorro de SRAM

Sin PROGMEM: 25 bytes en SRAM. Con PROGMEM: 0 bytes en SRAM, 25 bytes en Flash (32 KB disponibles). Siempre usar PROGMEM para tablas de solo lectura.

Actividad 3: Función find_active_rules()

// Estructura para almacenar una regla activa
typedef struct {
    uint8_t i_e;     // indice del termino de error
    uint8_t i_de;    // indice del termino de cambio
    uint8_t out_idx; // indice de salida (0..4)
    float   alpha;   // fuerza de disparo
} ActiveRule;

// Evaluar todas las reglas y retornar las activas
uint8_t find_active_rules(const FuzzyDegrees& mu_e,
                          const FuzzyDegrees& mu_de,
                          ActiveRule* rules_out,
                          uint8_t max_rules) {
    uint8_t count = 0;
    for (uint8_t i = 0; i < N_TERMS && count < max_rules; i++) {
        if (mu_e.mu[i] < 0.001f) continue; // skip inactiva
        for (uint8_t j = 0; j < N_TERMS && count < max_rules; j++) {
            if (mu_de.mu[j] < 0.001f) continue;
            float alpha = fmin2(mu_e.mu[i], mu_de.mu[j]);
            if (alpha > 0.0f) {
                rules_out[count].i_e     = i;
                rules_out[count].i_de    = j;
                rules_out[count].out_idx = get_rule(i, j);
                rules_out[count].alpha   = alpha;
                count++;
            }
        }
    }
    return count; // numero de reglas activas
}
